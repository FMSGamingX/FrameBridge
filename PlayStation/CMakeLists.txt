# FrameBridge • Faris Alsari / FMSGaming • Famous Gaming
cmake_minimum_required(VERSION 3.16)
project(FrameBridgePlayStation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(opt GUI CLI TESTS ANDROID BOREALIS SETSU STEAMDECK_NATIVE SPEEX PI_DECODER STEAM_SHORTCUT)
  set(CHIAKI_ENABLE_${opt} OFF CACHE STRING "" FORCE)
endforeach()
set(CHIAKI_USE_SYSTEM_CURL ON CACHE STRING "" FORCE)
set(CHIAKI_ENABLE_FFMPEG_DECODER ON CACHE STRING "" FORCE)
set(CHIAKI_USE_SYSTEM_NANOPB OFF CACHE STRING "" FORCE)
set(CHIAKI_USE_SYSTEM_JERASURE OFF CACHE STRING "" FORCE)
add_subdirectory("${CHIAKI_SOURCE}" chiaki EXCLUDE_FROM_ALL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SCALE REQUIRED IMPORTED_TARGET libswscale)
add_library(FrameBridge.PlayStation SHARED Client.cpp)
set_target_properties(FrameBridge.PlayStation PROPERTIES PREFIX "")
target_compile_definitions(FrameBridge.PlayStation PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(FrameBridge.PlayStation PRIVATE chiaki-lib PkgConfig::SCALE advapi32 xinput9_1_0 winmm)
add_executable(PS5DecoderTest EXCLUDE_FROM_ALL DecoderTest.cpp)
target_compile_definitions(PS5DecoderTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PS5DecoderTest PRIVATE chiaki-lib PkgConfig::SCALE advapi32 xinput9_1_0 winmm)
